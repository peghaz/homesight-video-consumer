cmake_minimum_required(VERSION 3.22)
project(HomesightCore VERSION 0.1.0 LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build")
endif()
option(HOMESIGHT_BUILD_PYTHON_BINDINGS "Build Python bindings" ON)
option(HOMESIGHT_ENABLE_S3_STORAGE "Enable S3-compatible storage" ON)
set(HOMESIGHT_SANITIZER "" CACHE STRING "Sanitizer: address,undefined or thread")
find_package(PkgConfig REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.20 gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-pbutils-1.0)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
    set(HOMESIGHT_SQLITE_TARGET SQLite3::SQLite3)
else()
    set(HOMESIGHT_SQLITE_TARGET SQLite::SQLite3)
endif()
find_package(Threads REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc video)
if(HOMESIGHT_ENABLE_S3_STORAGE)
    find_package(CURL 7.75 REQUIRED)
endif()
add_subdirectory(src)
if(HOMESIGHT_BUILD_PYTHON_BINDINGS)
    find_package(Python 3.12 REQUIRED COMPONENTS Interpreter Development.Module)
    find_package(pybind11 3.0.4 CONFIG REQUIRED)
    pybind11_add_module(_native python/bindings.cpp)
    target_link_libraries(_native PRIVATE HomesightCore::homesight_core)
    set_target_properties(_native PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
    install(TARGETS _native LIBRARY DESTINATION homesight_core)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(SKBUILD)
    set(HOMESIGHT_LIBRARY_INSTALL_DIR homesight_core/lib)
else()
    set(HOMESIGHT_LIBRARY_INSTALL_DIR ${CMAKE_INSTALL_LIBDIR})
endif()
install(TARGETS homesight_core EXPORT HomesightCoreTargets LIBRARY DESTINATION ${HOMESIGHT_LIBRARY_INSTALL_DIR} ARCHIVE DESTINATION ${HOMESIGHT_LIBRARY_INSTALL_DIR})
install(DIRECTORY include/homesight DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/HomesightCoreConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/HomesightCoreConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HomesightCore)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/HomesightCoreConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT HomesightCoreTargets NAMESPACE HomesightCore:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HomesightCore)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/HomesightCoreConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/HomesightCoreConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HomesightCore)
